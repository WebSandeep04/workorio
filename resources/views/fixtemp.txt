@push('scripts')
<script>
async function loadFields(){
    const list = document.getElementById('fb-field-list');
    list.innerHTML = '<div class="text-muted">Loading fields…</div>';
    try{
        const res = await fetch(`{{ route('formbuilder.fields') }}`);
        const data = await res.json();
        if (data.error){
            list.innerHTML = `<div class=\"text-danger\">${data.error}</div>`;
            return;
        }
        const cols = data.columns || [];
        const skip = ['id','created_at','updated_at','deleted_at'];
        list.innerHTML='';
        cols.forEach(col=>{
            if (skip.includes(col.name)) return;
            const id = `col_${col.name}`;
            const wrap = document.createElement('div');
            wrap.className='form-check';
            const cb = document.createElement('input');
            cb.type='checkbox';
            cb.className='form-check-input';
            cb.id=id;
            cb.dataset.name=col.name;
            cb.dataset.type=col.data_type;
            cb.checked = !!col.required;
            cb.addEventListener('change', buildPreview);
            const label = document.createElement('label');
            label.className='form-check-label';
            const req = col.required ? ' <span class="text-danger">*</span>' : '';
            label.setAttribute('for', id);
            label.innerHTML = `${col.name}${req} <small class=\"text-muted\">(${col.data_type})</small>`;
            wrap.appendChild(cb);
            wrap.appendChild(label);
            list.appendChild(wrap);
        });
        buildPreview();
    }catch(e){
        list.innerHTML = `<div class=\"text-danger\">${e.message}</div>`
    }
}

function toggleAll(state){
    document.querySelectorAll('#fb-field-list input[type="checkbox"]').forEach(cb=>{ cb.checked = state; });
    buildPreview();
}

function inferInputType(field){
    const name = (field.name||'').toLowerCase();
    const t = (field.data_type||'').toLowerCase();
    if (name.includes('email')) return 'email';
    if (name.includes('phone') || name.includes('mobile')) return 'tel';
    if (["int","bigint","integer","smallint","mediumint","tinyint","decimal","double","float"].includes(t)) return 'number';
    if (t === 'timestamp' || t === 'datetime') return 'datetime-local';
    if (t === 'date') return 'date';
    if (t.includes('text')) return 'textarea';
    return 'text';
}

function toTitle(str){ return (str||'').replaceAll('_',' ').replace(/\b\w/g, c=>c.toUpperCase()); }

function buildPreview(){
    const preview = document.getElementById('fb-preview');
    preview.innerHTML='';
    const form = document.createElement('form');
    form.className='row g-3';

    const selected = Array.from(document.querySelectorAll('#fb-field-list input[type="checkbox"]'))
        .filter(cb=>cb.checked)
        .map(cb=>({name: cb.dataset.name, data_type: cb.dataset.type}));

    selected.forEach(f=>{
        const group = document.createElement('div');
        group.className='col-12';
        const label = document.createElement('label');
        label.className='form-label fw-semibold';
        label.textContent = toTitle(f.name);
        let control;
        const t = inferInputType(f);
        if (t==='textarea'){
            control = document.createElement('textarea');
            control.className='form-control';
            control.rows=3;
        } else {
            control = document.createElement('input');
            control.type = t;
            control.className='form-control';
        }
        control.name = f.name;
        control.placeholder = `Enter ${toTitle(f.name)}`;
        group.appendChild(label);
        group.appendChild(control);
        form.appendChild(group);
    });

    const actions = document.createElement('div');
    actions.className='col-12 text-end';
    const btn = document.createElement('button');
    btn.type='submit';
    btn.className='btn btn-primary btn-sm';
    btn.textContent='Submit';
    actions.appendChild(btn);
    form.appendChild(actions);

    document.getElementById('fb-preview').appendChild(form);

    const tmp = document.createElement('div');
    tmp.appendChild(form.cloneNode(true));
    document.getElementById('fb-generated-html').value = prettyHtml(tmp.innerHTML);
}

function prettyHtml(html){
    const tab='  ';
    let out='';
    let indent=0;
    html.replace(/></g,'>$<').split('<').forEach(seg=>{
        if(!seg) return;
        if (seg.startsWith('/')) indent = Math.max(0, indent-1);
        out += `${' '.repeat(indent)}<${seg}\n`;
        if(!seg.startsWith('/') && !seg.endsWith('/>') && !seg.startsWith('input')){
            indent++;
        }
    });
    return out.trim();
}

// init
window.addEventListener('DOMContentLoaded', () => {
    loadFields();
    document.getElementById('fb-generate-btn').addEventListener('click', buildPreview);
    document.getElementById('fb-select-all').addEventListener('click', () => toggleAll(true));
    document.getElementById('fb-clear-all').addEventListener('click', () => toggleAll(false));
});
</script>
